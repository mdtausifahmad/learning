Skeleton

public sealed interface XEvent permits XEvent.Created, XEvent.Changed {
    record Created(String id, Instant at)               implements XEvent {}
    record Changed(String id, String value, Instant at) implements XEvent {}
}

public sealed interface XCommand permits XCommand.Create, XCommand.Change {
    record Create(String id)               implements XCommand {}
    record Change(String id, String value) implements XCommand {}
}

public sealed interface XState {            // only what decide() needs
    record Initial()           implements XState {}
    record Active(String id)   implements XState {}
}

public final class X {
    public static final XState INITIAL = new XState.Initial();
    private X() {}

    public static List<XEvent> decide(XCommand cmd, XState state, Clock clock) {
        var now = clock.instant();
        return switch (cmd) {
            case XCommand.Create c -> switch (state) {
                case XState.Initial i -> List.of(new XEvent.Created(c.id(), now));
                default -> throw new DomainException("already exists");
            };
            case XCommand.Change c -> {
                var a = requireActive(state);
                yield List.of(new XEvent.Changed(a.id(), c.value(), now));
            }
        };
    }

    public static XState evolve(XState state, XEvent event) {
        return switch (event) {               // exhaustive: no default
            case XEvent.Created e -> new XState.Active(e.id());
            case XEvent.Changed e -> state;   // nothing decide() needs changed
        };
    }

    public static XState fold(List<XEvent> history) {
        XState s = INITIAL;
        for (var e : history) s = evolve(s, e);
        return s;
    }

    private static XState.Active requireActive(XState s) {
        if (s instanceof XState.Active a) return a;
        throw new DomainException("not active");
    }
}

Given / When / Then test skeleton

class XTest {
    static final Clock CLOCK = Clock.fixed(Instant.parse("2026-01-01T00:00:00Z"), ZoneOffset.UTC);
    static final Instant T = CLOCK.instant();

    static List<XEvent> when(List<XEvent> given, XCommand cmd) {
        return X.decide(cmd, X.fold(given), CLOCK);
    }

    @Test void change_on_active_emits_Changed() {
        var given = List.<XEvent>of(new XEvent.Created("x-1", T));
        assertThat(when(given, new XCommand.Change("x-1", "v")))
            .containsExactly(new XEvent.Changed("x-1", "v", T));
    }

    @Test void create_twice_is_rejected() {
        var given = List.<XEvent>of(new XEvent.Created("x-1", T));
        assertThatThrownBy(() -> when(given, new XCommand.Create("x-1")))
            .isInstanceOf(DomainException.class);
    }
}

Review checklist

  • No I/O, Instant.now(), UUID.randomUUID() or Spring beans inside decide/evolve.
  • State has no display-only fields (names, lists, timestamps nobody checks).
  • No default -> on event switches; no validation in evolve.
  • Boundary drawn around an invariant; each command touches one stream.
  • One style per codebase: functional Decider or OO aggregate with uncommitted changes.
